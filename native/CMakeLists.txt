cmake_minimum_required(VERSION 3.16)
project(dnms_memory LANGUAGES CXX)
find_package(JNI REQUIRED)
add_library(dnms_memory SHARED memory_search.cpp)
target_include_directories(dnms_memory PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_features(dnms_memory PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(dnms_memory PRIVATE /O2)
else()
  target_compile_options(dnms_memory PRIVATE -O3)
endif()
